## 3. 测试数据与夹具

### 3.1 假 Docker stats 响应

```json
{"memory_stats": {"usage": 786432000, "limit": 1073741824,
  "stats": {"cache": 52428800, "inactive_file": 52428800}},
 "cpu_stats": {"cpu_usage": {"total_usage": 2000000000, "percpu_usage": [1, 1]},
   "system_cpu_usage": 8000000000, "online_cpus": 2},
 "precpu_stats": {"cpu_usage": {"total_usage": 1000000000}, "system_cpu_usage": 4000000000}}
```

`mem_mb = (786432000 - 52428800) / 2^20 = 700.0`。

### 3.2 构造已知斜率样本（回归断言）

```python
# 斜率 = +120 MB/h，跨度 2h，10 点，r2 ≈ 1.0
def linear_samples(slope_mb_per_hour: float, n: int = 10, span_sec: int = 7200,
                   start_mb: float = 400.0, t0: float = 1_700_000_000.0) -> list[dict]:
    step = span_sec / (n - 1)
    return [{"ts": t0 + i * step,
             "mb": round(start_mb + slope_mb_per_hour * (i * step) / 3600, 1),
             "cpu": 5.0} for i in range(n)]
```

锯齿样本（噪声，`r2 < 0.6`）：在均值附近交替 `±80MB`。

## 6. pytest 代码片段

```python
# tests/test_memory.py
import pytest
from probes.container_memory import ContainerMemoryProbe

def stats_builder(usage_mb=700, cache_mb=50, limit_mb=1024):
    return {"memory_stats": {"usage": int(usage_mb * 2**20), "limit": int(limit_mb * 2**20),
                             "stats": {"cache": int(cache_mb * 2**20)}},
            "cpu_stats": {"cpu_usage": {"total_usage": 2, "percpu_usage": [1, 1]},
                          "system_cpu_usage": 8, "online_cpus": 2},
            "precpu_stats": {"cpu_usage": {"total_usage": 1}, "system_cpu_usage": 4}}

def test_memory_cache_correction():
    used, cache, limit = ContainerMemoryProbe._memory(stats_builder(700, 50, 1024))
    assert (used, cache, limit) == (700.0, 50.0, 1024.0)   # C06-01

def linear_samples(slope, n=10, span=7200, start=400.0, t0=1_700_000_000.0):
    step = span / (n - 1)
    return [{"ts": t0 + i * step, "mb": round(start + slope * (i * step) / 3600, 1), "cpu": 5.0}
            for i in range(n)]

def test_slope_known_linear():
    slope, r2 = ContainerMemoryProbe._slope_mb_per_hour(linear_samples(120))
    assert abs(slope - 120) < 24 and r2 > 0.9           # C06-06

def sawtooth(n=10, span=7200, t0=1_700_000_000.0, mean=500.0, amp=80.0):
    step = span / (n - 1)
    return [{"ts": t0 + i * step, "mb": mean + (amp if i % 2 else -amp), "cpu": 5.0} for i in range(n)]

def test_sawtooth_is_noise():
    _, r2 = ContainerMemoryProbe._slope_mb_per_hour(sawtooth())
    assert r2 < 0.6                                      # C06-07

@pytest.mark.parametrize("mem,expect_crit", [(1099.9, False), (1100.0, True)])
@pytest.mark.asyncio
async def test_threshold_boundary(mem, expect_crit, probe_factory):
    p = probe_factory(stats=stats_builder(mem))
    r = await p.check()
    assert (r.status.value == "crit") is expect_crit     # C06-09/10
```
