## 6. pytest 代码片段

```python
# tests/test_probes_base.py
import asyncio, time, pytest
from probes.base import Probe
from core.models import ProbeResult, Status, Severity

class SlowProbe(Probe):
    name = "slow"
    async def _check(self) -> ProbeResult:
        await asyncio.sleep(10)
        return ProbeResult(self.name, self.target, Status.OK, Severity.INFO, "late", {}, time.time())

class BoomProbe(Probe):
    name = "boom"
    async def _check(self) -> ProbeResult:
        raise RuntimeError("boom")

@pytest.mark.asyncio
async def test_timeout_wrapped_as_unknown(deps_factory):
    deps = deps_factory(probe_timeout=0.1)
    r = await SlowProbe(cfg_default(), deps).check()
    assert r.status is Status.UNKNOWN and r.severity is Severity.WARN
    assert "超时" in r.message

@pytest.mark.asyncio
async def test_exception_wrapped_as_unknown(deps_factory):
    r = await BoomProbe(cfg_default(), deps_factory()).check()
    assert r.status is Status.UNKNOWN and "boom" in r.message
```

```python
# tests/test_policy.py
import pytest
from core.models import ProbeResult, Status, Severity
from core.policy import PolicyEngine

def mem(v: float) -> ProbeResult:
    return ProbeResult("container_memory", "napcat", Status.OK, Severity.INFO,
                       f"mem={v}", {"mem_mb": v}, 0.0)

@pytest.mark.parametrize("value,status,sev", [
    (699.9, Status.OK, Severity.INFO),
    (700.0, Status.WARN, Severity.WARN),      # >= warn 边界
    (700.1, Status.WARN, Severity.WARN),
    (1099.9, Status.WARN, Severity.WARN),
    (1100.0, Status.CRIT, Severity.CRIT),     # >= crit 边界
    (1100.1, Status.CRIT, Severity.CRIT),
])
def test_memory_thresholds(value, status, sev):
    f = PolicyEngine().evaluate([mem(value)])
    if status is Status.OK:
        assert f == []                        # OK 不产出 Finding
    else:
        assert len(f) == 1 and f[0].status is status and f[0].severity is sev
```

```python
# tests/test_bus.py
import pytest
from core.bus import EventBus

@pytest.mark.asyncio
async def test_subscriber_isolation():
    bus, seen = EventBus(), []
    async def good(t, p): seen.append(p)
    async def bad(t, p): raise RuntimeError("boom")
    bus.subscribe("scan.completed", bad)
    bus.subscribe("scan.completed", good)
    await bus.publish("scan.completed", {"n": 1})
    assert seen == [{"n": 1}]                 # 异常订阅者不影响正常订阅者
```
