## 1. 目标

1. 固化 `models.py` 六个契约类型，作为全项目唯一数据交换格式。
2. 建立「探针（取数）→ 判定（规则）→ 动作（副作用）」三段式流水线，判定纯函数化、动作可 mock（决策 D4）。

## 3. probes/base.py 完整代码

基类统一处理：**超时包装**、**异常兜底为 UNKNOWN**、**资源释放**；子类只实现 `_check()`。

```python
# probes/base.py
from __future__ import annotations
import asyncio
import time
from abc import ABC, abstractmethod
from typing import Any
from ..core.config import GuardianConfig
from ..core.models import ProbeResult, Severity, Status
class Probe(ABC):
    name: str = "probe"

    def __init__(self, config: GuardianConfig, deps: dict[str, Any], target: str = ""):
        self.config = config
        self.deps = deps
        self.target = target or self.name
        self.timeout = float(deps.get("probe_timeout", 8.0))

    @abstractmethod
    async def _check(self) -> ProbeResult:
        """子类实现：执行检查并返回 ProbeResult。"""

    async def check(self) -> ProbeResult:
        # 对外唯一入口：超时包装 + 异常兜底（决策 D10）
        try:
            return await asyncio.wait_for(self._check(), timeout=self.timeout)
        except asyncio.TimeoutError:
            return self._unknown(f"探针超时(>{self.timeout:.0f}s)")
        except Exception as exc:  # noqa: BLE001
            return self._unknown(f"探针异常: {exc!r}")

    def _unknown(self, message: str) -> ProbeResult:
        return ProbeResult(probe=self.name, target=self.target, status=Status.UNKNOWN,
                           severity=Severity.WARN, message=message, metrics={}, ts=time.time())

    async def close(self) -> None:   # 释放资源（连接/会话），按需覆写
        return None
```

## 4. 探针注册表 core/registry.py

模式通过**名字**引用探针，注册表把「名字 → 类」解耦，便于测试时替换为假探针。

```python
# core/registry.py
from __future__ import annotations
from typing import Callable
from ..probes.base import Probe
class ProbeRegistry:
    def __init__(self) -> None:
        self._factories: dict[str, type[Probe]] = {}

    def register(self, name: str, factory: type[Probe]) -> None:
        self._factories[name] = factory

    def get(self, name: str) -> type[Probe]:
        try:
            return self._factories[name]
        except KeyError as exc:
            raise KeyError(f"未注册的探针: {name}") from exc

registry = ProbeRegistry()

def register_probe(name: str) -> Callable[[type[Probe]], type[Probe]]:
    """类装饰器：@register_probe("qq_login")。"""
    def deco(cls: type[Probe]) -> type[Probe]:
        registry.register(name, cls)
        return cls
    return deco
```

## 5. PolicyEngine（core/policy.py）

判定输入 `list[ProbeResult]`，输出 `list[Finding]`。规则 `Rule(metric, op, threshold, status, severity, action_kind, reason)`；运算符支持 `> >= < <= == !=`；同一探针命中多条时取 **severity 最高**者；`UNKNOWN` 生成一条**无动作** Finding（只告警）。

```python
# core/policy.py
from __future__ import annotations
import operator
from dataclasses import dataclass
from typing import Any, Callable
from .models import ActionSpec, Finding, ProbeResult, Severity, Status
_OPS: dict[str, Callable[[Any, Any], bool]] = {
    ">": operator.gt, ">=": operator.ge, "<": operator.lt,
    "<=": operator.le, "==": operator.eq, "!=": operator.ne}
_RANK = {Severity.INFO: 0, Severity.WARN: 1, Severity.CRIT: 2, Severity.FATAL: 3}
@dataclass
class Rule:
    metric: str; op: str; threshold: float
    status: Status; severity: Severity
    action_kind: str | None = None; reason: str = ""

    def test(self, metrics: dict[str, Any]) -> bool:
        if self.metric not in metrics or self.op not in _OPS:
            return False
        try:
            return _OPS[self.op](float(metrics[self.metric]), float(self.threshold))
        except (TypeError, ValueError):
            return False
class PolicyEngine:
    """ProbeResult -> Finding，纯函数式，便于单测（决策 D4）。"""

    def __init__(self, rules: dict[str, list[Rule]] | None = None):
        self.rules = rules if rules is not None else DEFAULT_RULES

    def evaluate(self, results: list[ProbeResult]) -> list[Finding]:
        return [f for f in (self._evaluate_one(r) for r in results) if f is not None]

    def _evaluate_one(self, r: ProbeResult) -> Finding | None:
        if r.status == Status.UNKNOWN:
            return Finding(r.probe, r.target, r.status, r.severity, r.message, None)
        best: Rule | None = None
        for rule in self.rules.get(r.probe, []):
            if rule.test(r.metrics) and (best is None or _RANK[rule.severity] > _RANK[best.severity]):
                best = rule
        if best is None:
            return None if r.status == Status.OK else Finding(
                r.probe, r.target, r.status, r.severity, r.message, None)
        action = ActionSpec(kind=best.action_kind, target=r.target) if best.action_kind else None
        return Finding(r.probe, r.target, best.status, best.severity, best.reason or r.message, action)
# 默认规则（节选）；完整阈值见第 10 节探针清单
DEFAULT_RULES: dict[str, list[Rule]] = {
    "container_memory": [
        Rule("mem_mb", ">=", 1100, Status.CRIT, Severity.CRIT, "restart_container", "内存超严重阈值"),
        Rule("mem_mb", ">=", 700, Status.WARN, Severity.WARN, None, "内存超告警阈值")],
    "qq_login": [Rule("online", "==", 0, Status.CRIT, Severity.CRIT, "qq_relogin", "QQ 离线")],
}
```

## 6. HealthEngine.run_once 编排流程

流程：**并发跑探针 → 判定 → 逐条授权/执行/记录 → 发事件 → 通知**；全程异常隔离。

```python
# core/engine.py（节选：巡检编排；_probes/_actions/policy/safety/state/bus/notifier/deps 由 __init__ 注入）
import asyncio
import time
from uuid import uuid4
from astrbot.api import logger
from .models import Finding, Incident, ProbeResult, Severity, Status
class HealthEngine:
    async def run_once(self, trigger: str = "interval", scope: str = "all") -> list[Incident]:
        if self._running:                      # 幂等/重入保护（第 8 节）
            logger.warning("[guardian] 上一轮巡检未结束，跳过 trigger=%s", trigger)
            return []
        self._running = True
        try:
            probes = list(self._probes)        # scope 过滤见源码
            sem = asyncio.Semaphore(self.max_concurrency)

            async def guarded(p):              # 并发上限
                async with sem:
                    return await p.check()

            results = await asyncio.wait_for(  # 整体超时
                asyncio.gather(*(guarded(p) for p in probes), return_exceptions=True),
                timeout=self.overall_timeout)
            norm = [r if isinstance(r, ProbeResult) else self._unknown(p, r)   # 异常隔离
                    for p, r in zip(probes, results)]
            incidents = [await self._handle_finding(f) for f in self.policy.evaluate(norm)]
            await self.bus.publish("scan.completed", {"trigger": trigger, "count": len(incidents)})
            await self.notifier.dispatch(incidents)
            return incidents
        except asyncio.TimeoutError:
            logger.error("[guardian] 整体巡检超时(>%.0fs)，本轮结果丢弃", self.overall_timeout)
            return []
        finally:
            self._running = False
            self._last_scan_ts = time.time()

    async def _handle_finding(self, f: Finding) -> Incident:
        inc = Incident(id=uuid4().hex[:12], ts=time.time(), probe=f.probe, target=f.target,
                       status=f.status, severity=f.severity, message=f.reason,
                       action=f.suggested_action, result=None)
        if f.suggested_action is None:
            return await self._record(inc)
        verdict = self.safety.authorize(f.suggested_action, f.target, {"trigger": "scan"})
        action = self._actions.get(f.suggested_action.kind)
        if not verdict.allowed or action is None:
            inc.suppressed = True
            inc.suppress_reason = (verdict.reason if not verdict.allowed
                                   else f"无对应动作: {f.suggested_action.kind}")
            return await self._record(inc)
        f.suggested_action.dry_run = bool(self.config.dry_run)
        inc.result = await action.execute(f.suggested_action)
        return await self._record(inc)

    async def _record(self, inc: Incident) -> Incident:
        await self.state.record_incident(inc)     # B11
        self.deps["audit"].append(inc)            # B11：含被抑制的动作
        return inc

    def _unknown(self, p, exc) -> ProbeResult:
        return ProbeResult(probe=p.name, target=p.target, status=Status.UNKNOWN,
                           severity=Severity.WARN, message=f"探针异常: {exc!r}",
                           metrics={}, ts=time.time())
```

## 7. 并发与超时策略

| 机制 | 参数 | 实现 | 目的 |
| --- | --- | --- | --- |
| 单探针超时 | `deps["probe_timeout"]`，默认 8s | `Probe.check` 内 `asyncio.wait_for` | 单探针不拖垮整轮（R5） |
| 整体巡检超时 | `deps["overall_timeout"]`，默认 30s | `run_once` 外层 `wait_for` | 保证 tick 不重叠 |
| 并发上限 | `deps["max_concurrency"]`，默认 6 | `asyncio.Semaphore` 包裹 `check` | 避免瞬时打爆 Docker/HTTP（D10） |
| 异常隔离 / 资源释放 | — | `gather(return_exceptions=True)` / `terminate()` 逐个 `probe.close()` | 单探针异常转 UNKNOWN；释放会话 |

**规则**：任何阻塞调用（子进程、DNS）必须走 `to_thread`/`create_subprocess_exec`，禁止同步阻塞事件循环（D9）。

## 8. 幂等与重入保护

1. **单飞标志**：`self._running` 在 `run_once` 入口判断，正在跑则跳过、记日志、返回空列表。
2. **触发来源**：`trigger ∈ {interval, command, llm, startup}`；`command`/`llm` 触发被跳过时回执「上一轮巡检未结束，请稍后」。
3. **模式切换等待 / 热重载清理**：`switch_mode` 依赖 `await engine.wait_idle(timeout=30)`（轮询 `_running`，超时强制继续，见 B02 第 7 节）；`terminate()` 先 `scheduler.stop()` 再 `engine.close()`，确保旧实例不留定时器（R10）。

## 9. 事件总线 core/bus.py

进程内异步 pub/sub，解耦「探针/引擎」与「通知/审计/统计」；单个订阅者异常不影响其他订阅者。

```python
# core/bus.py
from __future__ import annotations
from collections import defaultdict
from typing import Any, Awaitable, Callable
Handler = Callable[[dict[str, Any]], Awaitable[None]]
class EventBus:
    def __init__(self) -> None:
        self._subs: dict[str, list[Handler]] = defaultdict(list)

    def subscribe(self, topic: str, handler: Handler) -> None:
        self._subs[topic].append(handler)

    async def publish(self, topic: str, payload: dict[str, Any]) -> None:
        for h in list(self._subs.get(topic, [])):
            try:
                await h(payload)
            except Exception:  # noqa: BLE001  单个订阅者异常隔离
                continue

    async def close(self) -> None:
        self._subs.clear()
```

**标准话题**：`scan.started`、`scan.completed`、`action.executed`、`action.suppressed`、`incident.recorded`、`profile.changed`、`mode.changed`。

## 10. 探针清单

| 模式 | 探针 | 检查项 | 关键指标 | 阈值/判据 | 建议动作 |
| --- | --- | --- | --- | --- | --- |
| server | `docker_daemon` | Docker Engine 可达性 | `reachable`, `api_version` | 不可达 → CRIT | 通知（无自动动作） |
| server | `container_health` | 容器状态与重启计数 | `running`, `restart_count`, `exit_code` | 未运行 → CRIT；重启计数骤增 → WARN | `start_container` |
| server | `container_memory` | 内存用量与增长速率 | `mem_mb`, `growth_mb_per_hour` | ≥`crit_mb`→CRIT；≥`warn_mb`→WARN；增速≥`growth_warn`→WARN | `restart_container`（仅 `restart_on_leak`） |
| server | `qq_login` | OneBot 在线状态 | `online`, `latency_ms` | 离线 → CRIT | `qq_relogin` |
| server | `self_health` | 插件自身健康 | `loop_lag_ms`, `task_count`, `last_scan_age` | 延迟/滞后超限 → WARN | 通知 |
| ros2 | `ros2_nodes` | 期望节点存在性 | `missing`, `nodes` | `missing ≥ 1` → CRIT | `ros2_restart` |
| ros2 | `ros2_topics` / `ros2_doctor` | 期望话题与频率 / `ros2 doctor` 汇总 | `missing`, `hz_min` / `error_count` | 缺失→CRIT；`hz<topic_hz_min`→WARN；有 error→WARN/CRIT | 通知 |

> 阈值最终以配置为准（`memory.warn_mb`/`crit_mb`/`growth_warn_mb_per_hour`、`ros2.topic_hz_min`）；表中数值为默认值。

## 11. 如何新增一个探针

1. 在 `probes/` 新建 `xxx_probe.py`，继承 `Probe`，设置 `name`，实现 `_check()`。
2. 用 `@register_probe("xxx")` 装饰类，使其可被注册表按名引用。
3. 在对应模式的 `build_probes()` 中加入实例（B02 第 6 节）。
4. 在 `core/policy.py` 的 `DEFAULT_RULES["xxx"]` 中补充规则（阈值/级别/建议动作），补单测（正常/超时/异常三态，C03），并在第 10 节探针清单登记。

```python
# probes/example_probe.py —— 模板
import time
from ..core.models import ProbeResult, Severity, Status
from ..core.registry import register_probe
from .base import Probe
@register_probe("example")
class ExampleProbe(Probe):
    name = "example"

    async def _check(self) -> ProbeResult:
        value = 42.0                              # 实际实现须异步、非阻塞取数
        ok = value < 100
        return ProbeResult(probe=self.name, target=self.target,
                           status=Status.OK if ok else Status.WARN,
                           severity=Severity.INFO if ok else Severity.WARN,
                           message=f"value={value}", metrics={"value": value}, ts=time.time())
```

## 12. 验收标准（对应 C03）

| 编号 | 判据 | 方法 |
| --- | --- | --- |
| C03-1 | 五个契约类型字段与 B00 第 6 节一致 | 契约单测 |
| C03-2 | 探针超时/内部异常均返回 `UNKNOWN` 且不抛异常 | 注入慢/异常探针 |
